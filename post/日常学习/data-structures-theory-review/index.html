一、时间复杂度分析（简答题 5 分）

核心考察：循环嵌套的复杂度推导，单循环中步长呈指数/对数变化的复杂度。

例题（2025 转专业真题）

int geti(int m) {
    for (int i = 1; i < m / 2;) {
        i = 2 * i;
    }
    return i;
}

答案：O(log⁡m)O(\log m)O(logm)。循环变量 iii 每次翻倍（i=2ii = 2ii=2i），从 111 增长到 m/2m/2m/2 需要 log⁡2(m/2)=log⁡2m−1\log_2(m/2) = \log_2 m - 1log2​(m/2)=log2​m−1 次迭代。

例题（2025 正考真题）

void func(int n) {
    int m = 0;
    for (int i = 1; i <= n; i++)
        for (int j = 1; j <= 2 * i; j++) m++;
}

答案：O(n2)O(n^2)O(n2)。内层循环执行 ∑i=1n2i=n(n+1)\sum_{i=1}^n 2i = n(n+1)∑i=1n​2i=n(n+1) 次，量级为 n2n^2n2。

二、链表（必考：简答 5 分 + 算法设计 15 分）

链表是期末考试高频考点。双链表的插入/删除操作在 2024 和 2025 连续两年出现在简答题中，而基于链表的算法设计也经常作为压轴题。

2.1 单链表与双链表的结构体定义

// 单链表节点
struct ListNode {
    int val;
    ListNode* next;
    ListNode(int x) : val(x), next(nullptr) {}
};

// 双链表节点
struct Node {
    int val;
    Node* prior;  // 前驱指针
    Node* next;   // 后继指针
};

2.2 单链表基本操作（常考代码）

头插法建立单链表

ListNode* head = nullptr;
for (int i = 0; i < n; i++) {
    ListNode* node = new ListNode(val);
    node->next = head;
    head = node;
}

尾插法建立单链表（保持输入顺序）

ListNode *head = nullptr, *tail = nullptr;
for (int i = 0; i < n; i++) {
    ListNode* node = new ListNode(val);
    if (!head) head = tail = node;
    else { tail->next = node; tail = node; }
}

单链表反转（经典三指针）

ListNode* reverseList(ListNode* head) {
    ListNode *prev = nullptr, *cur = head;
    while (cur) {
        ListNode* nxt = cur->next;  // 暂存下一个
        cur->next = prev;           // 反转指针
        prev = cur;                 // prev 前进
        cur = nxt;                  // cur 前进
    }
    return prev;  // 新头节点
}

单链表中删除指定值的所有节点

ListNode* removeElements(ListNode* head, int target) {
    ListNode dummy(0);                // 哑节点，统一处理头节点
    dummy.next = head;
    ListNode* p = &dummy;
    while (p->next) {
        if (p->next->val == target) {
            ListNode* del = p->next;
            p->next = p->next->next;  // 跨过被删节点
            delete del;
        } else {
            p = p->next;
        }
    }
    return dummy.next;
}

单链表在指定位置插入节点

// 在值为 x 的节点后面插入新节点 q
void insertAfter(ListNode* p, ListNode* q) {
    q->next = p->next;
    p->next = q;
}

// 在值为 x 的节点前面插入新节点 q（需要找到前驱）
void insertBefore(ListNode* head, ListNode* p, ListNode* q) {
    // 找 p 的前驱
    ListNode* prev = head;
    while (prev && prev->next != p) prev = prev->next;
    if (prev) {
        q->next = p;
        prev->next = q;
    }
}

单链表删除指定节点

// 删除 p 的后继节点
void deleteAfter(ListNode* p) {
    if (!p || !p->next) return;
    ListNode* del = p->next;
    p->next = p->next->next;
    delete del;
}

// 删除节点 p 本身（需要找前驱）
void deleteNode(ListNode* head, ListNode* p) {
    ListNode* prev = head;
    while (prev && prev->next != p) prev = prev->next;
    if (prev) {
        prev->next = p->next;
        delete p;
    }
}

2.3 双链表的插入与删除（简答题高频）

双链表的操作关键在于同时维护前驱和后继两个指针，操作顺序至关重要。

双链表在节点 p 后面插入新节点 q

操作前:  ... ⇄ p ⇄ p->next ⇄ ...
操作后:  ... ⇄ p ⇄ q ⇄ p->next(原) ⇄ ...
void insertAfter(Node* p, Node* q) {
    q->next = p->next;        // ① q 的后继 = p 原来的后继
    q->prior = p;             // ② q 的前驱 = p
    if (p->next)              // ③ 若 p 不是最后一个节点
        p->next->prior = q;   //    原后继的前驱指向 q
    p->next = q;              // ④ p 的后继指向 q
}

操作顺序为什么重要

先设置 q 的两个指针（①②），再修改周围节点指向 q（③④）。如果反过来先修改 p->next = q，就丢失了原后继的引用，无法再让原后继的前驱指向 q。

双链表在节点 p 前面插入新节点 q（2024 真题 / 2025 真题）

操作前:  ... ⇄ p->prior ⇄ p ⇄ ...
操作后:  ... ⇄ p->prior ⇄ q ⇄ p ⇄ ...
void insertBefore(Node* p, Node* q) {
    q->prior = p->prior;      // ① q 的前驱 = p 原来的前驱
    q->next = p;              // ② q 的后继 = p
    if (p->prior)             // ③ 若 p 不是第一个节点
        p->prior->next = q;   //    原前驱的后继指向 q
    p->prior = q;             // ④ p 的前驱指向 q
}

考试画图要求：画出操作前后的节点连接关系，用序号标注操作步骤。四个指针的修改顺序必须准确。

双链表删除节点 p

void deleteNode(Node* p) {
    if (!p) return;
    if (p->prior)              // p 有前驱
        p->prior->next = p->next;
    if (p->next)               // p 有后继
        p->next->prior = p->prior;
    delete p;
}

只需修改两个指针——让 p 的前驱和后继互相连接，跨过 p 即可。不需要遍历找到前驱，这是双链表相比单链表的优势。

双链表删除节点 p 的后继

void deleteAfter(Node* p) {
    if (!p || !p->next) return;   // p 为空或无后继
    Node* del = p->next;
    p->next = del->next;          // p 跨过 del
    if (del->next)                // del 有后继
        del->next->prior = p;     // 后继的前驱回指 p
    delete del;
}

双链表删除节点 p 的前驱

void deleteBefore(Node* p) {
    if (!p || !p->prior) return;   // p 为空或无前驱
    Node* del = p->prior;
    p->prior = del->prior;         // p 跨过 del
    if (del->prior)                // del 有前驱
        del->prior->next = p;      // 前驱的后继回指 p
    delete del;
}

// 遍历循环单链表
void traverse(ListNode* head) {
    if (!head) return;
    ListNode* p = head;
    do {
        cout << p->val << " ";
        p = p->next;
    } while (p != head);  // 绕一圈回到头
}

循环双链表：判断空表

// 空表：head->next == head && head->prior == head
bool isEmpty(Node* head) {
    return head->next == head && head->prior == head;
}

2.6 快慢指针技巧

// 找中间节点
ListNode* findMiddle(ListNode* head) {
    ListNode *slow = head, *fast = head;
    while (fast && fast->next) {
        slow = slow->next;
        fast = fast->next->next;
    }
    return slow;  // slow 停在中间(偶数个时停在后半段的开头)
}

// 判断链表是否有环（Floyd 判圈算法）
bool hasCycle(ListNode* head) {
    ListNode *slow = head, *fast = head;
    while (fast && fast->next) {
        slow = slow->next;
        fast = fast->next->next;
        if (slow == fast) return true;   // 快慢指针相遇 = 有环
    }
    return false;
}

// 找环的入口
ListNode* detectCycle(ListNode* head) {
    ListNode *slow = head, *fast = head;
    while (fast && fast->next) {
        slow = slow->next;
        fast = fast->next->next;
        if (slow == fast) {              // 相遇后
            ListNode* p = head;          // p 从头出发
            while (p != slow) {          // p 和 slow 同步走
                p = p->next;
                slow = slow->next;
            }
            return p;                    // 再次相遇点 = 环入口
        }
    }
    return nullptr;
}

2.7 单链表合并（有序）

// 合并两个升序单链表，结果仍为升序
ListNode* mergeTwoLists(ListNode* a, ListNode* b) {
    ListNode dummy(0);
    ListNode* tail = &dummy;
    while (a && b) {
        if (a->val <= b->val) { tail->next = a; a = a->next; }
        else                   { tail->next = b; b = b->next; }
        tail = tail->next;
    }
    tail->next = a ? a : b;  // 接上剩余部分
    return dummy.next;
}

三、栈与队列（简答题 5-10 分）

2.1 合法出栈序列

考点：给定入栈顺序，判断某个序列是否合法、或枚举所有可能的出栈序列。

判定方法：在任意时刻，已入栈但未出栈的元素必须满足「后进先出」——若某元素 xxx 已出栈，则所有在 xxx 之前入栈且尚未出栈的元素必须按入栈的逆序排列。

例题（2024 真题）：入栈顺序 A、B、C、D，找出以 B 开头的所有合法出栈序列。

答案：合法出栈序列 5 种：B A D C、B C D A、B C A D、B D C A、B A C D。

2.2 中缀转后缀

考点：手工转换中缀表达式为后缀表达式，以及辅助栈的深度。

规则：

  1. 操作数直接输出
  2. 运算符与栈顶比较优先级：当前 ≤\le≤ 栈顶时，弹出栈顶；再将当前压栈
  3. 左括号直接入栈，右括号弹出直到遇到左括号

例题（2025 真题）：将 (a+b∗c/d)∗e+f(a + b * c / d) * e + f(a+b∗c/d)∗e+f 转为后缀。问读到 / 时栈的内容。

步骤：

读 '(' : 栈 = (
读 'a' : 输出 a
读 '+' : 栈 = ( +
读 'b' : 输出 b
读 '*' : 栈 = ( + *
读 'c' : 输出 c
读 '/' : * 优先级 >= /，弹出 * → 输出 *；栈 = ( + ；再压入 / → 栈 = ( + /

答案：读到 / 时栈内容：( + /。后缀表达式：a b c * d / + e * f +

栈深度：该表达式中辅助栈最大深度为 444（( + * / 四者同时在栈时不会出现——最深为 ( + * 或 ( + /，深度为 333）。

3.2 已知遍历序列构建二叉树

必考题型：给定中序 + 后序（或中序 + 先序），画出二叉树并写出另一种遍历。

方法：后序最后一个/先序第一个 = 根，在中序中定位根后划分左右子树，递归构造。

例题（2025 真题）：中序 D B E A F C G，后序 D E B F G C A，画树并求先序。

解：

  • 后序最后一个 A 为根，中序 A 左边 D B E 为左子树，右边 F C G 为右子树
  • 左子树 {D,B,E} 的后序为 D E B（后序的前 3 个），根为 B
  • B 在中序中，左边 D、右边 E → B 的左子 D，右子 E
  • 右子树 {F,C,G} 的后序为 F G C，根为 C
  • C 在中序中，左边 F、右边 G → C 的左子 F，右子 G

先序：A B D E C F G

3.4 树、森林与二叉树的转换

森林 → 二叉树规则：

  1. 每棵树各自转为二叉树（左孩子右兄弟）
  2. 第一棵树的根作为总根，后续树的根依次作为前一棵树的右孩子

例题（2025 真题）：将给定森林转化为二叉树。

解：先将每棵树用「左孩子右兄弟」变成二叉树，再用右指针串联各树的根。

6.1 顺序查找与二分查找

顺序查找

nnn 个元素等概率查找：

ASL成功=n+12,ASL失败=n+1ASL_{成功} = \frac{n+1}{2}, \quad ASL_{失败} = n + 1ASL成功​=2n+1​,ASL失败​=n+1

适用链表或无序表，不需要元素有序。

二分查找判定树

二分查找过程可以表示为平衡二叉判定树。nnn 个元素时，树高 ⌈log⁡2(n+1)⌉\lceil \log_2(n+1) \rceil⌈log2​(n+1)⌉。

判定树性质：

  • 每次 mid 是判定树的根节点
  • 左子树 → target < a[mid]，右子树 → target > a[mid]
  • ASL 成功 = 所有成功结点深度之和 / n
  • ASL 失败 = 所有外部结点（失败结点）深度之和 / (n+1)

七、排序（简答题 5 分 + 应用题混合）

排序是期末考试固定出现的模块。考试不仅要求手工模拟每趟排序过程，还常考察各算法的特性对比（稳定性、时空复杂度、适用场景）。

7.1 插入排序类

直接插入排序

将未排序元素逐个插入到已排序部分的正确位置。基本有序时接近 O(n)O(n)O(n)——这是它的最大特点。

手工模拟（哨兵法，a[0]a[0]a[0] 为哨兵）：

初始: [49] 38 65 97 76 13 27
i=2:  38比49小 → [38 49] 65 97 76 13 27
i=3:  65比49大 → [38 49 65] 97 76 13 27
i=4:  97比65大 → [38 49 65 97] 76 13 27
i=5:  76插到97前 → [38 49 65 76 97] 13 27
i=6:  13挪到最前 → [13 38 49 65 76 97] 27
i=7:  27插到38前 → [13 27 38 49 65 76 97]

考试要点：

  • 稳定排序（相等元素不交换）
  • 最好 O(n)O(n)O(n)（正序），最坏 O(n2)O(n^2)O(n2)（逆序），平均 O(n2)O(n^2)O(n2)
  • 空间 O(1)O(1)O(1)
  • 适合数据量小或基本有序的场景

希尔排序

分组插入排序——将序列按增量 ddd 分成若干组，每组内做直接插入排序；逐步缩小 ddd 直到 d=1d=1d=1。

例题（2024 真题）：序列 49, 38, 65, 97, 76, 13, 27, 50，d=5,3,1d = 5, 3, 1d=5,3,1。

d=5d=5d=5（5 组）：

分组: (49,13) (38,27) (65,50) (97) (76)
每组排序后: 13, 27, 50, 97, 76, 49, 38, 65

d=3d=3d=3（3 组）：

分组: (13,97,38) (27,76,65) (50,49)
每组排序后: 13, 27, 49, 38, 65, 50, 97, 76

d=1d=1d=1（全序列插入排序）：

最终: 13, 27, 38, 49, 50, 65, 76, 97

考试要点：

  • 不稳定：相同元素可能分在不同组
  • 效率取决于增量序列的选择（如 Hibbard 序列 dk=2k−1d_k = 2^k - 1dk​=2k−1）
  • 最后一趟 d=1d=1d=1 就是普通的直接插入排序

7.2 交换排序类

冒泡排序

每趟将最大元素「浮」到最后。可以加标记优化：若某趟无交换，则已有序，提前结束。

初始: 49 38 65 97 76 13 27
第1趟: 38 49 65 76 13 27 [97]  (97就位)
第2趟: 38 49 65 13 27 [76 97]
第3趟: 38 49 13 27 [65 76 97]
第4趟: 38 13 27 [49 65 76 97]
第5趟: 13 27 [38 49 65 76 97]
第6趟: 13 [27 38 49 65 76 97]  (无交换, 提前结束)

考试要点：稳定、O(n2)O(n^2)O(n2)。最好 O(n)O(n)O(n)（已有序，一趟无交换即结束）。

快速排序（最重要的排序算法）

核心是分治：选基准 pivot，把比它小的放左边，比它大的放右边，再递归排序左右子区间。

例题：对 49, 38, 65, 97, 76, 13, 27 做快排，展示每趟结果。

初始: [49 38 65 97 76 13 27]
选49为pivot:
  从右找<49: 27 → 从左找>49: 65 → 交换27,65
  → [49 38 27 97 76 13 65]
  从右找<49: 13 → 从左找>49: 97 → 交换13,97
  → [49 38 27 13 76 97 65]
  左右指针相遇, 49与13交换
  → [13 38 27] 49 [76 97 65]   (一趟结束, 49归位)

左子区间[13 38 27], pivot=13:
  → 13 [38 27]

右子区间[76 97 65], pivot=76:
  → [65] 76 [97]

最终: 13 27 38 49 65 76 97

考试要点：

  • 不稳定，平均 O(nlog⁡n)O(n\log n)O(nlogn)，最坏 O(n2)O(n^2)O(n2)（正序或逆序时退化最严重——因为每次 pivot 都选到了边界元素）
  • 空间 O(log⁡n)O(\log n)O(logn)（递归栈深度）
  • 优化方法：随机选 pivot、三数取中法（避免最坏情况）

7.3 选择排序类

简单选择排序

每趟从未排序部分选最小元素放到已排序部分的末尾。

初始: [ ] 49 38 65 97 76 13 27
第1趟: [13] 38 65 97 76 49 27
第2趟: [13 27] 65 97 76 49 38
第3趟: [13 27 38] 97 76 49 65
第4趟: [13 27 38 49] 76 97 65
第5趟: [13 27 38 49 65] 97 76
第6趟: [13 27 38 49 65 76] 97

考试要点：不稳定（长距离交换可能打乱相等元素顺序），比较次数固定 n(n−1)/2n(n-1)/2n(n−1)/2，与初始排列无关。

堆排序

利用大根堆/小根堆进行选择排序。建堆 O(n)O(n)O(n) + n−1n-1n−1 次调整，每次 O(log⁡n)O(\log n)O(logn)，总 O(nlog⁡n)O(n\log n)O(nlogn)。

建堆过程（自底向上调整法）：从最后一个非叶节点 ⌊n/2⌋\lfloor n/2 \rfloor⌊n/2⌋ 开始，向前逐个调整。

堆排序步骤（以大根堆升序为例）：

  1. 建大根堆
  2. 堆顶（最大值）与末尾交换，堆大小减 1
  3. 堆顶向下调整（筛选）
  4. 重复 2-3 直到堆大小为 1

考试要点：不稳定，O(nlog⁡n)O(n\log n)O(nlogn) 且与初始排列无关。堆的定义（完全二叉树，父 ≥ 子女为大根堆）是常见判断题考点。

7.4 归并排序

将两个有序子序列合并为一个有序序列——分治，先递归分到单元素，再两两合并。

分: [49 38 65 97 76 13 27]
→ [49 38 65 97] [76 13 27]
→ [49 38] [65 97] [76 13] [27]
→ [49][38][65][97][76][13][27]

合(两两归并):
→ [38 49] [65 97] [13 76] [27]
→ [38 49 65 97] [13 27 76]
→ [13 27 38 49 65 76 97]

考试要点：稳定排序，O(nlog⁡n)O(n\log n)O(nlogn) 时间（与初始排列无关），空间 O(n)O(n)O(n)（需要辅助数组）。nnn 个元素做 kkk 路归并，趟数为 ⌈log⁡kn⌉\lceil \log_k n \rceil⌈logk​n⌉。

7.7 高频考点归纳

不稳定的四兄弟：希尔、快排、选择、堆排（口诀：希快选堆）

  • 每次堆顶出堆后的向下调整为 O(log⁡n)O(\log n)O(logn)，共 n−1n-1n−1 次调整

二路归并趟数：nnn 个元素，二路归并共 ⌈log⁡2n⌉\lceil \log_2 n \rceil⌈log2​n⌉ 趟

例题 2（2025 真题）：求二叉树中值为 ccc 的节点的祖先个数。

int findAncestors(TreeNode* root, char c) {
    if (!root) return -1;                     // 未找到
    if (root->val == c) return 0;             // 找到目标，祖先数为 0
    int left = findAncestors(root->left, c);
    if (left >= 0) return left + 1;           // 在左子树找到
    int right = findAncestors(root->right, c);
    if (right >= 0) return right + 1;         // 在右子树找到
    return -1;                                 // 两棵子树都没找到
}

例题 3（2025 真题）：求二叉树的宽度（LeetCode 662 定义：同一层最左和最右非空节点之间的跨度，包含中间的空节点位置）。

二叉树的宽度有两种常见定义，注意区分：

  1. 按节点数：每层节点数的最大值，直接 BFS 数即可
  2. 按跨度（LeetCode 662）：给每个节点分配堆式编号（根=1，左子=2i，右子=2i+1），每层宽度 = 最右编号 − 最左编号 + 1

2025 真题明确标注「可在 leetcode 662 提交」，采用的是第 2 种定义。

#include <queue>
using ull = unsigned long long;  // 深度深时编号可能溢出 int

int widthOfBinaryTree(TreeNode* root) {
    if (!root) return 0;
    queue<pair<TreeNode*, ull>> q;
    q.push({root, 1});
    ull maxWidth = 0;
    while (!q.empty()) {
        int sz = q.size();
        ull left = q.front().second;   // 本层最左编号
        ull right = left;
        for (int i = 0; i < sz; i++) {
            auto [node, idx] = q.front(); q.pop();
            right = idx;                // 逐次更新为最右编号
            if (node->left)  q.push({node->left,  idx * 2});
            if (node->right) q.push({node->right, idx * 2 + 1});
        }
        maxWidth = max(maxWidth, right - left + 1);
    }
    return maxWidth;
}

8.2 逆序对问题

例题（2025 真题）：求数组 AAA 中满足 i<ji < ji<j 且 A[i]>A[j]A[i] > A[j]A[i]>A[j] 的 (i,j)(i,j)(i,j) 对个数。要求 O(nlog⁡n)O(n\log n)O(nlogn)。

思路：归并排序过程中统计。当合并两个有序子数组时，若 left[i] > right[j]，则 left[i] 到 left[mid] 的所有元素都与 right[j] 构成逆序对，逆序对个数增加 mid - i + 1。

int merge(vector<int>& A, int L, int M, int R) {
    vector<int> tmp(R - L + 1);
    int i = L, j = M + 1, k = 0, cnt = 0;
    while (i <= M && j <= R) {
        if (A[i] <= A[j]) tmp[k++] = A[i++];
        else {
            cnt += M - i + 1;                 // 关键
            tmp[k++] = A[j++];
        }
    }
    while (i <= M) tmp[k++] = A[i++];
    while (j <= R) tmp[k++] = A[j++];
    for (int p = 0; p < k; p++) A[L + p] = tmp[p];
    return cnt;
}

int countInversions(vector<int>& A, int L, int R) {
    if (L >= R) return 0;
    int M = (L + R) / 2;
    return countInversions(A, L, M) + countInversions(A, M + 1, R) + merge(A, L, M, R);
}

8.4 胜者树（Winner Tree）

例题（2024 真题）：给定一串数字，构造胜者树。修改一个节点的值后，如何更新？

胜者树：完全二叉树，叶子存选手，内部节点存两个子节点的最小值的索引。类似淘汰赛，每层胜者向上晋级。根节点存储全局最小值。

更新步骤：从被修改的叶子节点出发，沿路径向上重新比较兄弟节点，更新每层的胜者，直到根。复杂度 O(log⁡n)O(\log n)O(logn)。

// 判断两棵胜者树是否相似
bool similarTree(TreeNode* t1, TreeNode* t2) {
    if (!t1 && !t2) return true;             // 都为空
    if (!t1 || !t2) return false;            // 一空一非空
    return similarTree(t1->left, t2->left)
        && similarTree(t1->right, t2->right);
}

8.6 链表算法设计（时空复杂度要求）

例题（2024 真题）：链表结构为 head -> ... -> 0 -> ...，000 之前有任意个节点，要求输出 000 前面的 nnn 个节点的值。O(m)O(m)O(m) 时间（mmm 为 0 之前的节点数），O(1)O(1)O(1) 空间。

思路：先遍历一趟找到 000 的位置（计数量 mmm），然后从头再次遍历，输出前 mmm 个节点的值。两趟都是 O(m)O(m)O(m)，只用了几个指针变量。

struct Node { int key, id; Node* next; };

void printBeforeZero(Node* head) {
    // 找 0 的位置
    Node* p = head, *zero = nullptr;
    while (p) {
        if (p->key == 0) { zero = p; break; }
        p = p->next;
    }
    // 输出 0 之前的数据
    p = head;
    while (p != zero) {
        cout << p->id << " ";
        p = p->next;
    }
}
